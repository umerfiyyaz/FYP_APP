// SCREEN - Trim video: user keeps only the part of the video they want analysed.
import { useEffect, useRef, useState } from "react";
import { PanResponder, Pressable, StyleSheet, Text, View } from "react-native";
import { useLocalSearchParams, useRouter } from "expo-router";
import { useVideoPlayer, VideoView } from "expo-video";
import { useSafeAreaInsets } from "react-native-safe-area-context";

import { Button } from "@/components/ui/Button";
import { Icon } from "@/components/ui/Icon";
import { setPendingUpload } from "@/services/pendingUpload";
import {
  colors,
  gradients,
  radius,
  shadows,
  spacing,
  type,
  withAlpha,
} from "@/theme";

const MIN_CLIP = 1; // seconds (same as the server)
const MAX_CLIP = 60; // seconds (same as max_clip_seconds on the server)
const HANDLE = 18;

const clamp = (v: number, lo: number, hi: number) =>
  Math.min(Math.max(v, lo), hi);
const round1 = (v: number) => Math.round(v * 10) / 10;
const fmt = (s: number) => {
  const t = Math.max(0, s);
  const m = Math.floor(t / 60);
  const sec = Math.floor(t % 60);
  return `${m}:${String(sec).padStart(2, "0")}`;
};

export default function TrimScreen() {
  const router = useRouter();
  const insets = useSafeAreaInsets();
  const params = useLocalSearchParams<{
    uri: string;
    name?: string;
    mime?: string;
    duration?: string;
  }>();

  const uri = String(params.uri ?? "");
  const fromPicker = Number(params.duration);
  const [duration, setDuration] = useState<number>(
    Number.isFinite(fromPicker) && fromPicker > 0 ? fromPicker : 0,
  );
  const [range, setRange] = useState({
    start: 0,
    end: Math.min(duration, MAX_CLIP),
  });
  const [pos, setPos] = useState(0);
  const [playing, setPlaying] = useState(false);

  // latest values for the gesture handlers (they are created once)
  const rangeRef = useRef(range);
  rangeRef.current = range;
  const durationRef = useRef(duration);
  durationRef.current = duration;
  const trackW = useRef(0);
  const dragBase = useRef(0);

  const player = useVideoPlayer(uri, (p) => {
    p.loop = false;
    p.timeUpdateEventInterval = 0.1;
  });

  // duration comes from the picker; if it was missing, read it from the video itself
  useEffect(() => {
    const subs = [
      player.addListener("sourceLoad", ({ duration: d }) => {
        if (durationRef.current <= 0 && d > 0) {
          setDuration(d);
          setRange({ start: 0, end: Math.min(d, MAX_CLIP) });
        }
      }),
      player.addListener("timeUpdate", ({ currentTime }) => {
        setPos(currentTime);
        const r = rangeRef.current;
        if (currentTime >= r.end) {
          player.pause();
          player.currentTime = r.start;
        }
      }),
      player.addListener("playingChange", ({ isPlaying }) =>
        setPlaying(isPlaying),
      ),
    ];
    return () => subs.forEach((s) => s.remove());
  }, [player]);

  useEffect(() => {
    if (duration > 0) setRange({ start: 0, end: Math.min(duration, MAX_CLIP) });
  }, [duration]);

  const makeResponder = (kind: "start" | "end") =>
    PanResponder.create({
      onStartShouldSetPanResponder: () => true,
      onPanResponderGrant: () => {
        player.pause();
        dragBase.current =
          kind === "start" ? rangeRef.current.start : rangeRef.current.end;
      },
      onPanResponderMove: (_, g) => {
        const w = trackW.current;
        const d = durationRef.current;
        if (!w || d <= 0) return;
        const delta = (g.dx / w) * d;
        const r = rangeRef.current;
        if (kind === "start") {
          const v = clamp(
            dragBase.current + delta,
            Math.max(0, r.end - MAX_CLIP),
            r.end - MIN_CLIP,
          );
          setRange({ start: v, end: r.end });
          player.currentTime = v;
        } else {
          const v = clamp(
            dragBase.current + delta,
            r.start + MIN_CLIP,
            Math.min(d, r.start + MAX_CLIP),
          );
          setRange({ start: r.start, end: v });
          player.currentTime = v;
        }
      },
      onPanResponderRelease: () => {
        player.currentTime = rangeRef.current.start;
      },
    });
  const startResponder = useRef(makeResponder("start")).current;
  const endResponder = useRef(makeResponder("end")).current;

  const togglePlay = () => {
    if (playing) {
      player.pause();
      return;
    }
    if (pos < range.start - 0.05 || pos >= range.end - 0.05)
      player.currentTime = range.start;
    player.play();
  };

  const reset = () => {
    player.pause();
    setRange({ start: 0, end: Math.min(duration, MAX_CLIP) });
    player.currentTime = 0;
  };

  const length = range.end - range.start;
  const tooShortVideo = duration > 0 && duration < MIN_CLIP;
  const isTrimmed = range.start > 0.05 || range.end < duration - 0.05;
  const leftPct = duration > 0 ? (range.start / duration) * 100 : 0;
  const rightPct = duration > 0 ? (range.end / duration) * 100 : 100;
  const playheadPct = duration > 0 ? clamp(pos / duration, 0, 1) * 100 : 0;

  const onContinue = () => {
    player.pause();
    setPendingUpload({
      uri,
      name: String(params.name ?? "video.mp4"),
      mimeType: String(params.mime ?? "video/mp4"),
      durationSec: duration,
      trimStart: isTrimmed ? round1(range.start) : null,
      trimEnd: isTrimmed ? round1(range.end) : null,
    });
    router.push("/processing");
  };

  return (
    <View style={[styles.container, { paddingTop: insets.top + spacing.sm }]}>
      {/* Header */}
      <View style={styles.header}>
        <Pressable
          onPress={() => router.back()}
          hitSlop={12}
          accessibilityLabel="Back"
        >
          <Icon name="arrow-back" size={24} color={colors.onSurface} />
        </Pressable>
        <Text style={[type.headlineSm, { color: colors.onSurface }]}>
          Trim video
        </Text>
        <Pressable onPress={reset} hitSlop={12} accessibilityLabel="Reset trim">
          <Text style={[type.labelMd, { color: colors.primary }]}>Reset</Text>
        </Pressable>
      </View>

      {/* Preview */}
      <View style={styles.previewWrap}>
        <VideoView
          player={player}
          style={styles.video}
          contentFit="contain"
          nativeControls={false}
        />
        <Pressable
          onPress={togglePlay}
          style={styles.playBtn}
          accessibilityLabel={playing ? "Pause" : "Play"}
        >
          <Icon
            name={playing ? "pause" : "play-arrow"}
            size={32}
            color={colors.onPrimary}
          />
        </Pressable>
      </View>

      {/* Numbers */}
      <View style={styles.statsRow}>
        <View style={styles.stat}>
          <Text style={[type.labelSm, styles.statLabel]}>START</Text>
          <Text style={[type.headlineSm, styles.statValue]}>
            {fmt(range.start)}
          </Text>
        </View>
        <View style={styles.stat}>
          <Text style={[type.labelSm, styles.statLabel]}>LENGTH</Text>
          <Text style={[type.headlineSm, styles.statValue]}>
            {length.toFixed(1)} s
          </Text>
        </View>
        <View style={styles.stat}>
          <Text style={[type.labelSm, styles.statLabel]}>END</Text>
          <Text style={[type.headlineSm, styles.statValue]}>
            {fmt(range.end)}
          </Text>
        </View>
      </View>

      {/* Trim bar */}
      <View
        style={styles.trackOuter}
        onLayout={(e) => {
          trackW.current = e.nativeEvent.layout.width;
        }}
      >
        <View style={styles.track} />
        {/* kept part */}
        <View
          pointerEvents="none"
          style={[
            styles.kept,
            {
              left: `${leftPct}%`,
              width: `${Math.max(rightPct - leftPct, 0)}%`,
            },
          ]}
        />
        {/* playhead */}
        <View
          pointerEvents="none"
          style={[styles.playhead, { left: `${playheadPct}%` }]}
        />
        {/* handles */}
        <View
          {...startResponder.panHandlers}
          hitSlop={{ top: 16, bottom: 16, left: 16, right: 16 }}
          style={[styles.handle, { left: `${leftPct}%` }]}
        />
        <View
          {...endResponder.panHandlers}
          hitSlop={{ top: 16, bottom: 16, left: 16, right: 16 }}
          style={[styles.handle, { left: `${rightPct}%` }]}
        />
      </View>

      <View style={styles.scaleRow}>
        <Text style={[type.labelSm, styles.statLabel]}>0:00</Text>
        <Text style={[type.labelSm, styles.statLabel]}>{fmt(duration)}</Text>
      </View>

      <Text style={[type.bodySm, styles.hint]}>
        {tooShortVideo
          ? "This video is too short. Please choose another one."
          : duration > MAX_CLIP
            ? `Drag the yellow handles. You can keep up to ${MAX_CLIP} seconds.`
            : "Drag the yellow handles to keep only the part you want analysed."}
      </Text>

      {/* Actions */}
      <View
        style={[styles.footer, { paddingBottom: insets.bottom + spacing.md }]}
      >
        <Button
          label={isTrimmed ? "Continue with trimmed clip" : "Continue"}
          icon="check"
          iconPosition="left"
          onPress={onContinue}
          gradient={gradients.ctaFade}
          paddingVertical={15}
          shadow={shadows.cta}
          disabled={duration <= 0 || tooShortVideo}
        />
      </View>
    </View>
  );
}

// Holds the video the user picked + the part they kept, until the upload starts.
// (Processing screen reads this when we connect it to the backend.)

export type PendingUpload = {
  uri: string;
  name: string;
  mimeType: string;
  durationSec: number;
  /** null = whole video (no trimming) */
  trimStart: number | null;
  trimEnd: number | null;
};

let pending: PendingUpload | null = null;

export const setPendingUpload = (p: PendingUpload | null) => {
  pending = p;
};
export const getPendingUpload = () => pending;
export const clearPendingUpload = () => {
  pending = null;
};

function stageFromPercent(p: number) {
  if (p < 25) return "preparing";
  if (p < 45) return "finding_cat";
  if (p < 70) return "reading_behavior";
  if (p < 90) return "listening_sound";
  return "health_report";
}

export default function ProcessingScreen() {
  const router = useRouter();
  const insets = useSafeAreaInsets();

  const cat = useActiveCat();
  const [percent, setPercent] = useState(USE_MOCK ? 64 : 3);
  const [phraseIdx, setPhraseIdx] = useState(0);
  const [stage, setStage] = useState<string>("preparing");
  const [failure, setFailure] = useState<string | null>(null);
  const [attempt, setAttempt] = useState(0);
  const analysisId = useRef<string | null>(null);
  const userCancelled = useRef(false);
  const job = useRef<{
    attempt: number;
    promise: Promise<UploadResponse>;
  } | null>(null);

  // Progressive simulation timer (demo mode only)
  useEffect(() => {
    if (!USE_MOCK) return;
    const timer = setInterval(() => {
      setPercent((prev) => {
        if (prev >= 98) {
          clearInterval(timer);
          return 98;
        }
        const next = prev + 1;
        if (next % 6 === 0) {
          setPhraseIdx((idx) => (idx + 1) % 4);
        }
        return next;
      });
    }, 800);

    return () => clearInterval(timer);
  }, []);

  // Real mode: upload the video, then ask the server for the progress until it is finished.
  useEffect(() => {
    if (USE_MOCK) return;
    const pending = getPendingUpload();
    if (!cat || !pending) {
      setFailure(
        "There is no video to analyse. Please choose a video from your gallery first.",
      );
      return;
    }
    const catId = cat.id;
    let cancelled = false;
    let timer: ReturnType<typeof setTimeout> | undefined;
    setFailure(null);
    setPercent(3);
    setStage("preparing");

    const finish = async (id: string) => {
      await Promise.all([
        historyActions.hydrate(catId),
        catActions.hydrate(),
        alertActions.hydrate(catId),
      ]);
      clearPendingUpload();
      if (!cancelled) router.replace(`/results/${id}` as never);
    };

    const poll = async (id: string, errors: number) => {
      if (cancelled) return;
      try {
        const s = await api<{
          status: string;
          progress: number;
          stage: string | null;
          error: string | null;
        }>(`/analyses/${id}/status`);
        if (cancelled) return;
        setPercent(Math.max(8, Math.min(100, s.progress)));
        if (s.stage) setStage(s.stage);
        if (s.status === "done") {
          finish(id);
          return;
        }
        if (s.status === "no_cat") {
          clearPendingUpload();
          router.replace("/no-cat-found");
          return;
        }
        if (s.status === "failed") {
          setFailure(s.error ?? "The analysis failed. Please try again.");
          return;
        }
        timer = setTimeout(() => poll(id, 0), 1500);
      } catch {
        if (cancelled) return;
        if (errors >= 3) {
          setFailure(
            "Lost the connection to the server. Check your Wi-Fi and try again.",
          );
          return;
        }
        timer = setTimeout(() => poll(id, errors + 1), 2000);
      }
    };

    // one upload per attempt (also when React runs this effect twice in development)
    if (!job.current || job.current.attempt !== attempt) {
      job.current = { attempt, promise: uploadVideo(catId, pending) };
    }
    job.current.promise
      .then((up) => {
        if (cancelled) {
          // the user left while the video was uploading: remove what the server just created
          if (userCancelled.current)
            historyApi.remove(up.analysis_id).catch(() => {});
          return;
        }
        analysisId.current = up.analysis_id;
        setPercent(8);
        poll(up.analysis_id, 0);
      })
      .catch((e) => {
        if (!cancelled) setFailure(errorMessage(e));
      });

    return () => {
      cancelled = true;
      if (timer) clearTimeout(timer);
    };
  }, [attempt]);

  // Real mode: rotate the mascot sentence every few seconds.
  useEffect(() => {
    if (USE_MOCK) return;
    const t = setInterval(() => setPhraseIdx((i) => (i + 1) % 4), 4000);
    return () => clearInterval(t);
  }, []);

  const pendingNow = getPendingUpload();
  const clipSec = pendingNow
    ? pendingNow.trimStart !== null && pendingNow.trimEnd !== null
      ? pendingNow.trimEnd - pendingNow.trimStart
      : pendingNow.durationSec
    : 0;
  const activeStage = USE_MOCK ? stageFromPercent(percent) : stage;
  const activeIdx = Math.max(
    0,
    STEPS.findIndex((s) => s.key === activeStage),
  );

  const totalCircumference = 440;
  const strokeDashoffset =
    totalCircumference - (totalCircumference * percent) / 100;
  const remainingSeconds = Math.max(
    2,
    Math.round(8 * (1 - (percent - 64) / 34)),
  );

  const onCancel = () => {
    Alert.alert(
      "Cancel Analysis",
      "Are you sure you want to cancel the analysis?",
      [
        { text: "Continue Analysis", style: "cancel" },
        {
          text: "Yes, Cancel",
          style: "destructive",
          onPress: () => {
            userCancelled.current = true;
            const id = analysisId.current;
            if (!USE_MOCK && id) historyApi.remove(id).catch(() => {}); // stop it on the server too
            clearPendingUpload();
            router.back();
          },
        },
      ],
    );
  };

  return (
    <View style={styles.container}>
      {/* Top Header */}
      <View style={[styles.header, { paddingTop: Math.max(insets.top, 12) }]}>
        <View style={styles.headerLeft}>
          <PressScale
            onPress={() => router.back()}
            accessibilityRole="button"
            accessibilityLabel="Back"
            style={styles.headerIconBtn}
          >
            <Icon name="arrow-back" size={22} color={colors.onSurface} />
          </PressScale>
          <Image
            source={images.logoSmall}
            style={styles.headerLogo}
            resizeMode="contain"
          />
          <Text style={[type.headlineMd, styles.headerTitle]}>Results</Text>
        </View>

        <View style={styles.headerRight}>
          <PressScale
            accessibilityRole="button"
            accessibilityLabel="Share"
            style={styles.headerIconBtn}
          >
            <Icon name="share" size={20} color={colors.onSurface} />
          </PressScale>
          <LinearGradient
            colors={[colors.primaryContainer, colors.secondary]}
            style={styles.avatarBorder}
          >
            <Image
              source={images.userProfile}
              style={styles.profileAvatar}
              resizeMode="cover"
            />
          </LinearGradient>
        </View>
      </View>

      <ScrollView
        style={styles.scroll}
        contentContainerStyle={styles.scrollContent}
        showsVerticalScrollIndicator={false}
      >
        {/* Radial Ambient Glows */}
        <Glow
          size={240}
          color={colors.primaryContainer}
          opacity={0.12}
          style={{ top: 0, alignSelf: "center" }}
        />
        <Glow
          size={180}
          color={colors.tertiaryContainer}
          opacity={0.08}
          style={{ top: 120, alignSelf: "center" }}
        />

        {/* Circular Progress Gauge */}
        <View style={styles.gaugeSection}>
          <View style={styles.circleSvgWrap}>
            <Svg width={180} height={180} viewBox="0 0 160 160">
              <Circle
                cx={80}
                cy={80}
                r={70}
                fill="none"
                stroke={colors.surfaceContainerHighest}
                strokeWidth={8}
                opacity={0.45}
              />
              <Circle
                cx={80}
                cy={80}
                r={70}
                fill="none"
                stroke={colors.primaryContainer}
                strokeWidth={9}
                strokeDasharray={`${totalCircumference} ${totalCircumference}`}
                strokeDashoffset={strokeDashoffset}
                strokeLinecap="round"
                originX={80}
                originY={80}
                rotation={-90}
              />
            </Svg>

            <View style={styles.gaugeCenterLabel}>
              <View style={styles.percentRow}>
                <Text style={[type.headlineXl, styles.percentNumber]}>
                  {percent}
                </Text>
                <Text style={[type.headlineMd, styles.percentSign]}>%</Text>
              </View>
              <View style={styles.aiActiveBadge}>
                <Pulse duration={1000} minOpacity={0.3}>
                  <View style={styles.activeDot} />
                </Pulse>
                <Text style={[type.labelSm, styles.aiActiveText]}>
                  AI ACTIVE
                </Text>
              </View>
            </View>
          </View>

          <Text style={[type.headlineSm, styles.timeLeftText]}>
            {USE_MOCK
              ? `About ${remainingSeconds} seconds left`
              : "Analysing your clip"}
          </Text>
          <Text style={[type.bodyMd, styles.gaugeSubtitle]}>
            {USE_MOCK
              ? "We analyse about 30 seconds of the best cat footage."
              : "This can take a little while. You can keep this screen open."}
          </Text>

          {USE_MOCK || clipSec > 0 ? (
            <View style={styles.analysedPill}>
              <Icon name="timer" size={15} color={colors.secondary} />
              <Text style={[type.labelMd, { color: colors.onSurface }]}>
                {USE_MOCK ? (
                  <>
                    Seconds analysed:{" "}

                      30
                    </Text>{" "}
                    of 48
                  </>
                ) : (
                  <>
                    Clip length:{" "}
                    <Text style={{ color: colors.primary, fontWeight: "700" }}>
                      {Math.round(clipSec)} s
                    </Text>
                  </>
                )}
              </Text>
            </View>
          ) : null}
        </View>

        {failure ? (
          <View
            style={{
              borderRadius: radius.lg,
              padding: spacing.md,
              gap: spacing.sm,
              backgroundColor: withAlpha(colors.errorContainer, 0.35),
            }}
          >
            <View
              style={{ flexDirection: "row", alignItems: "center", gap: 8 }}
            >
              <Icon name="warning" size={18} color={colors.error} />
              <Text style={[type.labelLg, { color: colors.error }]}>
                Could not finish
              </Text>
            </View>
            <Text style={[type.bodyMd, { color: colors.onSurface }]}>
              {failure}
            </Text>
            {getPendingUpload() ? (
              <Button
                label="Try again"
                icon="refresh"
                iconPosition="left"
                onPress={() => setAttempt((a) => a + 1)}
                gradient={gradients.ctaFade}
                paddingVertical={13}
                shadow={shadows.cta}
              />
            ) : null}
            <Button
              label="Back"
              onPress={() => router.back()}
              color={colors.onSurface}
              background={withAlpha(colors.surfaceContainerHigh, 0.8)}
              paddingVertical={13}
              shadow={shadows.md}
            />
          </View>
        ) : null}

        {/* Neural Analysis Matrix Card */}
        <View style={styles.matrixCard}>
          <View style={styles.matrixHeader}>
            <View style={styles.matrixHeaderLeft}>
              <View style={styles.matrixIconCircle}>
                <Icon name="psychology" size={16} color={colors.primary} />
              </View>
              <Text style={[type.headlineSm, { color: colors.onSurface }]}>
                Neural Analysis Matrix
              </Text>
            </View>
            {USE_MOCK ? (
              <View style={styles.liveVersionBadge}>
                <Text style={[type.labelSm, { color: colors.primary }]}>
                  v4.2 Live
                </Text>
              </View>
            ) : null}
          </View>

          {/* Steps List */}
          <View style={styles.stepsList}>
            {STEPS.map((st, i) => (
              <StepRow
                key={st.key}
                title={st.title}
                sub={st.sub}
                state={
                  i < activeIdx
                    ? "done"
                    : i === activeIdx
                      ? "running"
                      : i === activeIdx + 1
                        ? "waiting"
                        : "queued"
                }
              />
            ))}

            {/* Step 6: Saved to history */}
            <View style={styles.stepRowQueued}>
              <View style={styles.stepLeftStart}>
                <View style={styles.queuedIconCircle}>
                  <Icon
                    name="save-as"
                    size={16}
                    color={colors.onSurfaceVariant}
                  />
                </View>
                <View style={{ flex: 1 }}>
                  <Text
                    style={[type.bodyMd, { color: colors.onSurfaceVariant }]}
                  >
                    Saved to history
                  </Text>
                  <Text
                    style={[
                      type.labelSm,
                      { color: withAlpha(colors.onSurfaceVariant, 0.8) },
                    ]}
                  >
                    When finished, automatically saved to history.
                  </Text>
                </View>
              </View>
              <View style={styles.waitingTag}>
                <Text
                  style={[type.labelSm, { color: colors.onSurfaceVariant }]}
                >
                  Queued
                </Text>
              </View>
            </View>
          </View>
        </View>

        {/* Mascot Sense Speech Card */}
        <View style={styles.mascotSpeechCard}>
          <View style={styles.mascotAvatarCircle}>
            <Image
              source={images.mascotSense}
              style={styles.mascotAvatarImg}
              resizeMode="cover"
            />
          </View>
          <View style={styles.speechBubble}>
            <View style={styles.speechHeader}>
              <Text style={[type.labelSm, styles.mascotSenseTitle]}>
                MASCOT SENSE
              </Text>
              <View style={styles.mintDot} />
            </View>
            <Text numberOfLines={1} style={[type.bodySm, styles.speechText]}>
              {mascotPhrases(cat?.name ?? "your cat")[phraseIdx % 4]}
            </Text>
          </View>
        </View>

        {/* Action: Cancel Button */}
        <View style={styles.cancelWrap}>
          <PressScale
            onPress={onCancel}
            accessibilityRole="button"
            accessibilityLabel="Cancel analysis"
            style={styles.cancelBtn}
          >
            <Icon name="close" size={18} color={colors.error} />
            <Text style={[type.labelLg, { color: colors.onSurface }]}>
              Cancel
            </Text>
          </PressScale>
        </View>
      </ScrollView>
    </View>
  );
}
